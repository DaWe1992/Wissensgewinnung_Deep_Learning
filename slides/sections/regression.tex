% SUBSECTION: Wiederholung: Lineare Regression
\subsection{Wiederholung: Lineare Regression}

% Lineare Regression
\begin{frame}{Lineare Regression}{}
	Wir wiederholen die wichtigsten Aussagen zur \keyword{linearen Regression}:

	\begin{itemize}
		\item Wir betrachten einen Datensatz bestehend aus $N$ Featurevektoren $\bphi(\bx_n) \in \R^M, n = 1, 2, \ldots, N$.
		\item Die Funktion $\bphi$ ist hierbei eine \keyword{Basisfunktion}, z.\,B. die polynomiale Basis.
		\item Um den \keyword{Bias} zu modellieren, vereinbaren wir, dass stets $\phi_1(\bx) = 1$ gilt.
		\item Des Weiteren ist jeder Vektor $\bphi(\bx_n) \in \R^M$ mit einem zugehörigen Label $y_n \in \R$ versehen, $n = 1, 2, \ldots, N$.
		\item Das lineare Regressionsmodell hat die Form
		\begin{equation*}
			h(\bx; \btheta) = \sum_{m=1}^M \theta_m \phi_m(\bx) = \btheta^\top \bphi(\bx).
		\end{equation*}
	\end{itemize}
\end{frame}


% Design Matrix
\begin{frame}{Design Matrix}{}
	\begin{itemize}
		\item Wir fassen die $N$ Featurevektoren $\bphi(\bx_n) \in \R^M$ zeilenweise zu der Matrix $\bPhi$ zusammen.
		\item Diese hat damit die allgemeine Form
		\begin{equation*}
			\bPhi \coloneqq
			\begin{pmatrix}
				\horzbar 	& \bphi(\bx_1) 	& \horzbar 	\\
				\horzbar 	& \bphi(\bx_2) 	& \horzbar 	\\
				\vdots 	&\vdots 		& \vdots 		\\
				\horzbar 	& \bphi(\bx_N) 	& \horzbar
			\end{pmatrix} \in \R^{N \times M}.
		\end{equation*}
		\item Die erste Spalte ist hierbei konstant 1 \textit{(wegen der Dummy-Komponente für den Bias)}.
		\item Wir nennen diese Matrix die \keyword{Design Matrix}.
		\item Analog sammeln wir alle Labels in einem Vektor
		\begin{equation*}
			\by \coloneqq \bigl( y_1, y_2, \ldots, y_N \bigr)^\top \in \R^N.
		\end{equation*}
	\end{itemize}
\end{frame}


% Methode der kleinsten Quadrate
\begin{frame}{Methode der kleinsten Quadrate}{}
	\begin{itemize}
		\item Der Vektor $\btheta \in \R^M$ beinhaltet die \keyword{Modellparameter}.
		\item Diese möchten wir so wählen, dass unser Modell die Daten möglichst gut erklärt
		\item Die Abweichung der Vorhersagen zu den Trainingsdaten sollte also \textit{möglichst klein} sein.
	\end{itemize}

	\begin{mydef}[def:reg-least-squares-error]
		Mit den obigen Definitionen ist der \keyword{Least-Squares-Fehler} gegeben durch
		\begin{equation*}
			\JLS(\btheta)
				\coloneqq \frac{1}{2N} \Vert \bPhi \btheta - \by \Vert^2
				= \frac{1}{2N} \bigl( \bPhi \btheta - \by \bigr)^\top \bigl( \bPhi \btheta - \by \bigr).
		\end{equation*}
	\end{mydef}

	\begin{boxGray}
		\textbf{Bemerkung:} Der Vektor $\bPhi \btheta \in \R^N$ beinhaltet die \textbf{Modellvorhersagen},
		wenn man dem Modell die Parameter $\btheta$ zugrunde legt.
		$\JLS$ misst also die \textbf{mittlere quadratische Abweichung} der Vorhersagen zu den wahren Werten $\by$.
	\end{boxGray}
\end{frame}


% Minimierung der Fehlerfunktion
\begin{frame}{Minimierung der Fehlerfunktion}{}
	Unser Ziel ist die Minimierung von $\JLS$. Hierzu schreiben $\JLS$ unter Ausnutzung der Transpositionsregeln
	$(\bA + \bB)^\top = \bA^\top + \bB^\top$ und $(\bA \bB)^\top = \bB^\top \bA^\top$ zunächst um:
	\begin{align*}
		\JLS(\btheta)
			&= \frac{1}{2N} \Vert \bPhi \btheta - \by \Vert^2
				= \frac{1}{2N} \bigl( \bPhi \btheta - \by \bigr)^\top \bigl( \bPhi \btheta - \by \bigr)
	\\[2mm]
			&= \frac{1}{2N} \bigl( (\bPhi \btheta)^\top - \by^\top \bigr) \bigl( \bPhi \btheta - \by \bigr)
	\\[2mm]
			&= \frac{1}{2N} \bigl( (\bPhi \btheta)^\top \bPhi \btheta - (\bPhi \btheta)^\top \by - \by^\top (\bPhi \btheta) + \by^\top \by \bigr)
	\\[2mm]
			&= \frac{1}{2N} \bigl( \btheta^\top \bPhi^\top \bPhi \btheta - 2 (\bPhi \btheta)^\top \by + \by^\top \by \bigr)
	\\[2mm]
			&= \frac{1}{2N} \bigl( \btheta^\top \bPhi^\top \bPhi \btheta - 2 \btheta^\top \bPhi^\top \by + \by^\top \by \bigr)
	\end{align*}
\end{frame}


% Minimierung der Fehlerfunktion (Forts.)
\begin{frame}{Minimierung der Fehlerfunktion (Forts.)}{}
	Um einen stationären Punkt zu finden, müssen wir $\JLS$ nach $\btheta$ ableiten und Null setzen.

	\begin{itemize}
		\item Es ist $\frac{\partial}{\partial \btheta} \btheta^\top \bPhi^\top \bPhi \btheta = 2 \bPhi^\top \bPhi \btheta$,
			da $\bPhi^\top \bPhi$ symmetrisch ist. \\ \vspace*{2mm}
		\begin{myproof}
			Wir nutzen die Transpositionsregeln und erhalten $\bigl( \bPhi^\top \bPhi \bigr)^\top =
			\bPhi^\top \bigl( \bPhi^\top \bigr)^\top = \bPhi^\top \bPhi$, woraus die Symmetrie folgt.
		\end{myproof} \vspace*{2mm}
		\item Es ist $\frac{\partial}{\partial \btheta} \bigl( -2 \btheta^\top \bPhi^\top \by \bigr) = -2 \bPhi^\top \by$
			und $\frac{\partial}{\partial \btheta} \by^\top \by = \0$.
		\item Der Gradient der Fehlerfunktion $\JLS$ ist damit gegeben durch
		\begin{equation*}
			\frac{\partial}{\partial \btheta} \JLS(\btheta)
				= \frac{1}{2 N} \bigl( 2 \bPhi^\top \bPhi \btheta - 2 \bPhi^\top \by \bigr)
				\propto \bPhi^\top \bPhi \btheta - \bPhi^\top \by
				= \bPhi^\top \bigl( \bPhi \btheta - \by \bigr)
		\end{equation*}
	\end{itemize}
	
	\begin{boxGray}
		\textbf{Bemerkung:} Man beachte den Zusammenhang zwischen dem Gradienten und dem Vorhersagefehler:
		Je größer der Fehler, desto größer auch der Gradient!
	\end{boxGray}
\end{frame}


% Minimierung der Fehlerfunktion (Forts.)
\begin{frame}{Minimierung der Fehlerfunktion (Forts.)}{}
	Wir setzen den Gradienten Null und lösen nach $\btheta$ auf:
	\begin{alignat*}{3}
		\frac{\partial}{\partial \btheta} \JLS(\btheta) \overset{!}{=} \0
			&\quad\Longleftrightarrow\quad
			&\bPhi^\top \bPhi \btheta - \bPhi^\top \by
			&= \0
	\\[2mm]
			&\quad\Longleftrightarrow\quad
			&\bPhi^\top \bPhi \btheta
			&= \bPhi^\top \by
	\\[2mm]
			&\quad\Longleftrightarrow\quad
			&\btheta^\star
			&= \bigl( \bPhi^\top \bPhi \bigr)^{-1} \bPhi^\top \by
	\end{alignat*}

	\begin{boxGray}
		\textbf{Bemerkung:} Wir haben damit einen stationären Punkt von $\JLS$ gefunden. Allerdings wissen wir noch nicht,
		ob es sich tatsächlich um ein Minimum handelt. Außerdem sind wir hier etwas naiv von der Existenz der Inversen
		$\bigl( \bPhi^\top \bPhi \bigr)^{-1}$ ausgegangen. Existiert diese Matrix immer? \\
		\textit{Mit diesen Fragen beschäftigen wir uns im Folgenden.}
	\end{boxGray}
\end{frame}


% Ein paar nützliche Aussagen
\begin{frame}{Ein paar nützliche Aussagen}{}
	\begin{mytheo}[th:reg-A-top-A-psd]
		Für alle $\bA \in \R^{N \times M}$ ist $\bA^\top \bA$ positiv semidefinit.
	\end{mytheo}
	
	\begin{myproof}
		Für alle $\bz \in \R^M$ gilt
		$\bz^\top \bigl( \bA^\top \bA \bigr) \bz = (\bA \bz)^\top (\bA \bz) = \Vert \bA \bz \Vert^2 \ge 0$,
		woraus die Behauptung folgt.	
	\end{myproof}

	\begin{mytheo}[th:reg-col-rank-pd]
		Für alle $\bA \in \R^{N \times M}$ mit \textit{vollem Spaltenrang} ist $\bA^\top \bA$ positiv definit.
	\end{mytheo}

	\begin{myproof}
		Da $\bA$ vollen Spaltenrang hat, sind die Spaltenvektoren linear unabhängig.
		Daher ist $\bA \bz$ für alle $\bz \in \R^M \backslash \{ \0 \}$ eine nicht-triviale Linearkombination
		der Spalten von $\bA$. Folglich gilt $\bA \bz \ne \0$ und wir erhalten
		$$\bz^\top \bigl( \bA^\top \bA \bigr) \bz = (\bA \bz)^\top (\bA \bz) = \Vert \bA \bz \Vert^2 > 0,$$ woraus die Behauptung folgt.
	\end{myproof}
\end{frame}


% Ein paar nützliche Aussagen (Forts.)
\begin{frame}{Ein paar nützliche Aussagen (Forts.)}{}
	\begin{mytheo}[th:reg-col-rank-inv]
		Wenn $\bA \in \R^{N \times N}$ positiv definit ist, so ist $\bA$ auch invertierbar.
	\end{mytheo}
	
	\begin{myproof}
		Wir nehmen an, $\bA$ sei nicht invertierbar. Dann sind die Spalten von $\bA$ linear abhängig und es existiert ein
		$\bz \in \R^N \backslash \{ \0 \}$ mit $\bA \bz = \0$.
		Daraus folgt aber $$\bz^\top \bA \bz = \bz^\top \0 = 0,$$ was der vorausgesetzten positiven Definitheit von $\bA$ widerspricht.
		Es muss $\bA$ folglich invertierbar sein.
	\end{myproof}
\end{frame}


% Handelt es sich nun um ein Minimum?
\begin{frame}{Handelt es sich nun um ein Minimum?}{}
	Um diese Frage zu klären, berechnen wir die \keyword{Hesse-Matrix} von $\JLS$ und untersuchen,
	ob die Matrix im stationären Punkt $\btheta^\star$ positiv definit ist.

	\begin{itemize}
		\item Man rechnet nach, dass die Hesse-Matrix von $\JLS$ gegeben ist durch
			$$\frac{\partial^2}{\partial \btheta \partial \btheta} \JLS(\btheta) = \bPhi^\top \bPhi.$$
		\item Mit Satz \ref{th:reg-col-rank-pd} ist diese Matrix positiv definit für alle $\btheta$, falls $\bPhi$ vollen Spaltenrang hat,
			d.\,h. \textit{die Features müssen linear unabhängig sein}.
			In diesem Fall handelt es sich bei $\btheta^\star$ tatsächlich um ein Minimum.
		\item Unter derselben Voraussetzung haben wir damit sogar die \keyword{Konvexität} von $\JLS$ bewiesen \textbf{(warum?)},
			sodass das Minimum \textit{global und eindeutig} ist.
	\end{itemize}
\end{frame}


% Und wie sieht es nun mit der Existenz der Inversen aus?
\begin{frame}{Und wie sieht es nun mit der Existenz der Inversen aus?}{}
	\begin{itemize}
		\item Mit Satz \ref{th:reg-col-rank-inv} ist die Existenz der Inversen sichergestellt, wenn $\bPhi^\top \bPhi$ positiv definit ist,
			was wiederum aus der linearen Unabhängigkeit der Features folgt (Satz \ref{th:reg-col-rank-pd}).
		\item \textbf{Achtung:} Bei der numerischen Berechnung der Inversen auf dem Computer kann es
			trotzdem zu Problemen kommen.
		\item Es ist \textit{numerisch stabiler}, die Inverse nicht explizit zu berechnen,
			sondern stattdessen das folgende lineare Gleichungssystem in $\btheta$ zu lösen
			\begin{equation*}
				\bPhi^\top \bPhi \btheta = \bPhi^\top \by.
			\end{equation*}
		\item Alternativ kann das Optimierungsproblem mit dem \keyword{Gradientenabstiegsverfahren} gelöst werden.
		\item Iteriere hierzu die Regel:
		\begin{equation}
			\btheta_{t+1} \longleftarrow \btheta_t - \alpha \bPhi^\top \bigl( \bPhi \btheta_t - \by \bigr). 
		\end{equation}
	\end{itemize}
\end{frame}


% Normalengleichungen
\begin{frame}{Normalengleichungen}{}
	Wir fassen unsere Ergebnisse zusammen:
	
	\begin{mytheo}[th:reg-normal-equations]
		Die Design Matrix $\bPhi \in \R^{N \times M}$ des Regressionsproblems habe vollen Spaltenrang,
		d.\,h. die Features seien linear unabhängig.
		Dann existiert die Inverse $\bigl( \bPhi^\top \bPhi \bigr)^{-1}$ und
		\begin{equation*}
			\btheta^\star = \bigl( \bPhi^\top \bPhi \bigr)^{-1} \bPhi^\top \by
		\end{equation*}
		ist das eindeutige globale Minimum von $\JLS$.
		Die Gleichungen $\bPhi^\top \bPhi \btheta = \bPhi^\top \by$ bezeichnet man als \keyword{Normalengleichungen}.
	\end{mytheo}
\end{frame}


% Probabilistische Regression
\subsection{Probabilistische Regression}

% Eine probabilistische Sicht
\begin{frame}{Eine probabilistische Sicht}{}
	\begin{boxGray}
		\textbf{Annahme 1:} Die Zielwerte $y \in \R$ und die Features $\bphi(\bx) \in \R^M$ hängen über die Gleichung
		\begin{equation}
			y = \btheta^\top \bphi(\bx) + \varepsilon
		\end{equation}
		zusammen. Der Fehlerterm $\varepsilon \in \R$ repräsentiert nicht modellierte Effekte oder Datenrauschen.
	\end{boxGray}
	
	\begin{boxGray}
		\textbf{Annahme 2:} Für den Fehlerterm gilt $\varepsilon \sim \norm\bigl( 0, \sigma^2 \bigr)$.
	\end{boxGray}

	\begin{boxGray}
		\textbf{Bemerkung:} Der Zielwert $y$ kann damit als Zufallsvariable aufgefasst werden, für die gilt
		\begin{equation}
			y \mid \bx \sim \norm\bigl(\btheta^\top \bphi(\bx), \sigma^2 \bigr).
		\end{equation}
	\end{boxGray}
\end{frame}

% Maximum-Likelihood Regression
\begin{frame}{Maximum-Likelihood Regression}{}
	\begin{itemize}
		\item Wir betrachten wieder einen Datensatz $\bigl\{ (\bx_n, y_n) \bigr\}_{n=1}^N$ bestehend aus $N$ Trainingsbeispielen.
		\item Die Likelihood Funktion des Modells ist gegeben durch
		\begin{equation}
			\label{eq:reg-likelihood}
			\begin{aligned}
				L(\btheta, \sigma^2)
					&= p\bigl( \by \mid \bX; \btheta, \sigma^2 \bigr)
			\\[2mm]
					&= \prod_{n=1}^N p\bigl( y_n \mid \bx_n; \btheta, \sigma^2 \bigr)
			\\[2mm]
					&= \prod_{n=1}^N \frac{1}{\sqrt{2 \pi \sigma^2}} \cdot \exp\left(
						-\frac{1}{2 \sigma^2} \biggl( y_n - \btheta^\top \bphi(\bx_n) \biggr)^2
					\right)
			\end{aligned}
		\end{equation}
	\end{itemize}
\end{frame}


% Maximum-Likelihood Regression (Forts.)
\begin{frame}{Maximum-Likelihood Regression (Forts.)}{}
	\begin{itemize}
		\item Durch Logarithmieren von \eqref{eq:reg-likelihood} erhalten wir die Log-Likelihood Funktion
		\begin{equation}
			\label{eq:reg-log-likelihood}
			\begin{aligned}
				l\bigl( \btheta, \sigma^2 \bigr)
					&\coloneqq \ln p\bigl( \by \mid \bX; \btheta, \sigma^2 \bigr)
			\\[2mm]
					&= -\frac{N}{2} \ln\bigl( 2 \pi \sigma^2 \bigr)
						- \frac{1}{2 \sigma^2} \sum_{n=1}^N \bigl( y_n - \btheta^\top \bphi(\bx_n) \bigr)^2.
			\end{aligned}
		\end{equation}
		\item Aus \eqref{eq:reg-log-likelihood} wird ersichtlich, dass der Least-Squares-Fehler minimiert werden muss, um die
			Likelihood in $\btheta$ zu maximieren.
		\item Die Lösung für $\btheta$ ist dieselbe wie in Satz \ref{th:reg-normal-equations}, zusätzlich kann aber die \textit{globale
			Unsicherheit} des Modells quantifiziert werden
		\vspace*{-3mm}
		\begin{equation}
			\bthetaestML = \bigl( \bPhi^\top \bPhi \bigr)^{-1} \bPhi^\top \by
		\qquad\qquad
			\sigmasquML = \frac{1}{N} \sum_{n=1}^N \bigl( y_n - \bthetaestML^\top \bphi(\bx_n) \bigr)^2.
		\end{equation}
	\end{itemize}
\end{frame}

% Maximum Aposteriori (MAP) Regression
\subsection{Maximum Aposteriori (MAP) Regression}

% Berücksichtigung einer Apriori-Verteilung
\begin{frame}{Berücksichtigung einer Apriori-Verteilung}{}
	\begin{itemize}
		\item Die Maximum-Likelihood Lösung ist \textbf{anfällig für Overfitting}.
		\item Eine Möglichkeit der \keyword{Regularisierung} ist die Verwendung einer \keyword{Apriori-Verteilung} \textit{(Prior)}.
		\item Für den Prior des Parametervektors $\btheta$ benutzen wir die Normalverteilung
		\begin{equation}
			\btheta \sim \norm\bigl( \0, b^2 \bI \bigr).
		\end{equation}
		\item Hierbei kontrolliert $b > 0$ die Varianz der Parameter.
		\item Es handelt sich hierbei um eine \keyword{isotropische Normalverteilung} \textit{(rotationsinvariant)}.
		\item Der Prior kann benutzt werden, um Vorwissen zu berücksichtigen: Er kodiert, welche Parameterwerte apriori plausibel sind.
		\item Häufig ist der Prior jedoch \keyword{nicht informativ} (z.\,B. Gleichverteilung) und wird eher aus mathematischer Bequemlichkeit gewählt.
	\end{itemize}
\end{frame}


% Bestimmung des Posteriors
\begin{frame}{Bestimmung des Posteriors}{}
	\begin{itemize}
		\item Um den Posterior zu bestimmen, benutzen wir den \keyword{Satz von Bayes}
		\begin{equation} \label{eq:reg-bayes-theorem}
			\begin{aligned}
				p\bigl( \btheta \mid \bX, \by \bigr)
					&= \frac{\overbracket{
						p\bigl( \by \mid \bX, \btheta; \sigma^2 \bigr)
					}^{\text{Likelihood}} \cdot \overbracket{p\bigl( \btheta; b^2 \bigr)}^{\text{Prior}}}{p\bigl( \by \mid \bX \bigr)}
			\\[2mm]
					&\propto p\bigl( \by \mid \bX, \btheta; \sigma^2 \bigr) \cdot p\bigl( \btheta; b^2 \bigr).
			\end{aligned}
		\end{equation}
		\item Wir betrachten $\btheta$ nun als (vektorwertige) Zufallsvariable \textit{(sie steht nun vor dem Semikolon)}.
		\item Bisher haben wir nur die Likelihood maximiert, nun berücksichtigen wir auch den Prior $p\bigl( \btheta; b^2 \bigr)$.
	\end{itemize}
	
	\begin{boxGray}
		\textbf{Frage:} Warum ignorieren wir den Nenner in \eqref{eq:reg-bayes-theorem}?
	\end{boxGray}
\end{frame}


% Optimierung des Posteriors
\begin{frame}{Optimierung des Posteriors}{}
	\begin{itemize}
		\item Anstelle der Log-Likelihood maximieren wir nun den \keyword{Log-Posterior}.
		\item Diesen erhalten wir aus \eqref{eq:reg-bayes-theorem} durch Logarithmieren
		\begin{equation} \label{eq:reg-log-posterior}
			\ln p\bigl( \btheta \mid \bX, \by \bigr)
				= \ln p\bigl( \by \mid \bX, \btheta; \sigma^2 \bigr) + \ln p\bigl( \btheta; b^2 \bigr) + \const.
		\end{equation}
		\item Die Konstante '$\const$' enthält alle Terme, die nicht von $\btheta$ abhängen.
		\item Die \keyword{MAP-Schätzung} \textit{(Maximum Aposteriori)} ist ein Kompromiss zwischen dem daten\underline{un}abhängigen Prior
			und der datenabhängigen Likelihood.
		\item Wir betrachten im Folgenden den \keyword{negativen Log-Posterior} indem wir \eqref{eq:reg-log-posterior} negieren und erhalten
			für den MAP-Schätzer
		\begin{equation} \label{eq:reg-neg-log-posterior}
			\bthetaestMAP \coloneqq \argmin_{\btheta}\Bigl\{ -\ln p\bigl( \by \mid \bX, \btheta; \sigma^2 \bigr)
				- \ln p\bigl( \btheta; b^2 \bigr) \Bigr\}.
		\end{equation}
	\end{itemize}
\end{frame}


% Optimierung des Posteriors (Forts.)
\begin{frame}{Optimierung des Posteriors (Forts.)}{}
	Unter Verwendung der Transpositionsregeln schreiben wir \eqref{eq:reg-neg-log-posterior} um zu
	\begin{equation} \label{eq:reg-neg-log-posterior-new}
		\begin{aligned}
			-\ln p\bigl( \btheta \mid \bX, \by \bigr)
				&= -\ln p\bigl( \by \mid \bX, \btheta; \sigma^2 \bigr) - \ln p\bigl( \btheta; b^2 \bigr) + \const
		\\[2mm]
				&= \frac{1}{2 \sigma^2} \bigl( \by - \bPhi \btheta \bigr)^\top \bigl( \by - \bPhi \btheta \bigr)
					+ \frac{1}{2 b^2} \btheta^\top \btheta + \const
		\\[2mm]
				&= \frac{1}{2 \sigma^2} \bigl( \by^\top \by - \by^\top \bPhi \btheta - (\bPhi \btheta)^\top \by
					+ (\bPhi \btheta)^\top \bPhi \btheta \bigr) + \frac{1}{2 b^2} \btheta^\top \btheta + \const
		\\[2mm]
				&= \frac{1}{2 \sigma^2} \bigl( \btheta^\top \bPhi^\top \bPhi \btheta - 2 \btheta^\top \bPhi^\top \by \bigr)
					+ \frac{1}{2 b^2} \btheta^\top \btheta + \const.
		\end{aligned}
	\end{equation}
	
	\begin{mytask}[task:reg-log-posterior]
		Überlegen Sie sich, wie man in \eqref{eq:reg-neg-log-posterior-new} von Zeile 1 auf Zeile 2 kommt.
	\end{mytask}
\end{frame}


% Aufgabe: Berechnung des Gradienten
\begin{frame}{Aufgabe: Berechnung des Gradienten}{}
	\begin{mytask}[task:reg-log-posterior-gradient]
		Jetzt sind Sie an der Reihe:
		\begin{enumerate}
			\item Berechnen Sie den Gradienten des negativen Log-Posteriors $-\ln p\bigl( \btheta \mid \bX, \by \bigr)$.
				Benutzen Sie hierzu die Form aus \eqref{eq:reg-neg-log-posterior-new}.
			\item Setzen Sie den Gradienten anschließend Null und lösen Sie nach $\btheta$ auf.

				Zeigen Sie, dass diese Inverse \textit{(zumindest analytisch)} immer existiert.
		\end{enumerate}
	\end{mytask}
\end{frame}


% Vergleich der MLE und MAP Lösungen
\begin{frame}{Vergleich der MLE und MAP Lösungen}{}
	\begin{boxGray}
		\textbf{MLE-Schätzer:}
		\begin{equation}
			\bthetaestML = \bigl( \bPhi^\top \bPhi \bigr)^{-1} \bPhi^\top \by
		\end{equation}
	\end{boxGray}
	
	\begin{boxGray}
		\textbf{MAP-Schätzer:}
		\begin{equation}
			\bthetaestMAP = \left( \bPhi^\top \bPhi + \frac{\sigma^2}{b^2} \bI \right)^{-1} \bPhi^\top \by
		\end{equation}
		
		\textbf{Bemerkung:} Setzt man $\lambda \coloneqq \sigma^2 / b^2$,
		so entspricht die MAP-Schätzung der Ridge-Regression.
	\end{boxGray}
\end{frame}

% Bayes'sche Regression
\subsection{Bayes'sche Regression}

% Bayes'sche Regression
\begin{frame}{Bayes'sche Regression}{}
	\begin{itemize}
		\item Wir haben gesehen, dass MLE-Schätzungen zu \keyword{Overfitting} neigen.
		\item Für die MAP-Schätzung haben wir deshalb zusätzlich einen Prior $p(\btheta)$ berücksichtigt,
			welcher der \keyword{Regularisierung} dient.
	\end{itemize}
	
	\begin{boxGray}
		Die \keyword{Bayes'sche Regression} erweitert diesen Ansatz folgendermaßen:

		\begin{itemize}
			\item Wieder benutzen wir einen Prior $p(\btheta)$ für die Modellparameter $\btheta$.
			\item \textbf{Aber:} Wir berechnen diesmal keine Punktschätzung der Parameter.
			\item Stattdessen nutzen wir die \textit{volle Posterior-Verteilung} aus, um Vorhersagen zu machen.
			\item Hierzu berechnet man die \keyword{Predictive Distribution}, die sämtliche Parameterwerte mit
				ihren zugehörigen Wahrscheinlichkeiten in die Vorhersage einfließen lässt.
		\end{itemize}
	\end{boxGray}
\end{frame}


% Modellannahmen
\begin{frame}{Modellannahmen}{}
	\begin{itemize}
		\item Für den Prior benutzen wir wieder eine Normalverteilung
		\begin{equation}
			\btheta \sim \norm\bigl( \bmu_0, \bSigma_0 \bigr).
		\end{equation}
		\item Es ist $\bmu_0$ der Mittelwert des Priors (oft benutzt man $\bmu_0 \coloneqq \0$) und $\bSigma_0$ dessen Kovarianzmatrix
			(wie oben ist diese häufig von der Form $\bSigma_0 \coloneqq b^2 \bI$ mit $b > 0$).
		\item Die Likelihood modellieren wir ebenfalls analog zu oben
		\begin{equation}
			y \mid \newinst, \btheta \sim \norm\bigl( \btheta^\top \bphi(\newinst), \sigma^2 \bigr).
		\end{equation}
		\item Hierbei ist $\newinst$ ein Datenpunkt, für den eine Vorhersage berechnet werden soll.
	\end{itemize}
\end{frame}


% Prior Predictive Distribution
\begin{frame}{Prior Predictive Distribution}{}
	\begin{itemize}
		\item In der Praxis interessieren uns die konkreten Werte der Parameter in der Regel nicht.
		\item Wir wollen hauptsächlich die Labels für neue Instanzen vorhersagen.
		\item Sei also $\newinst$ eine Testinstanz.
		\item Die \keyword{Prior Predictive Distribution} ist gegeben durch
		\begin{equation} \label{eq:reg-prior-predictive-dist}
			p(y \mid \newinst)
				= \int_{\btheta} p(y \mid \newinst, \btheta) \cdot p(\btheta)
				= \E_{\btheta}\bigl[ p(y \mid \newinst, \btheta) \bigr].
		\end{equation}
		\item Man kann \eqref{eq:reg-prior-predictive-dist} als \textit{mittlere Vorhersage} interpretieren.
		\item Es handelt sich um einen \textit{gewichteten Durchschnitt über alle möglichen Parameterwerte gemäß ihrer Wahrscheinlichkeit.}
	\end{itemize}
\end{frame}


% Aufgabe: Bestimmung der Verbunddichte
\begin{frame}{Aufgabe: Bestimmung der Verbunddichte}{}
	\begin{mytask}[task:reg-bayes-reg-joint-dist]
		Der Integrand in \eqref{eq:reg-prior-predictive-dist} ist die Verbunddichte von $y$ und $\btheta$. Mit der Produktregel gilt
		\begin{equation*}
			p\bigl( y \mid \newinst, \btheta; \sigma^2 \bigr) \cdot p\bigl( \btheta; b^2 \bigr) = p\bigl( y, \btheta \mid \newinst; \sigma^2, b^2 \bigr).
		\end{equation*}
		Bestimmen Sie einen Ausdruck für diese Verbunddichte.
		
		\textbf{Ein paar Hinweise:}
		\begin{itemize}
			\item Sie müssen Erwartungswert und Kovarianz der Verbundverteilung bestimmen,
				denn ein Produkt von Normalverteilungen ist wieder normalverteilt.
			\item Schreiben Sie $y$ in der Form $y = \btheta^\top \bphi(\newinst) + \varepsilon$ mit
				$\varepsilon \sim \norm\bigl( 0, \sigma^2 \bigr)$.
			\item Verwenden Sie die Rechenregeln für den Erwartungswert, insbesondere die Linearität.
		\end{itemize}
	\end{mytask}
\end{frame}


% Parameter der Prior Predictive Distribution
\begin{frame}{Parameter der Prior Predictive Distribution}{}
	Die Parameter der Predictive Distribution lassen sich analytisch bestimmen:
	
	\begin{mytheo}[th:reg-prior-predictive-dist]
		Für die Prior Predictive Distribution gilt
		\begin{equation*}
			y \mid \newinst \sim \norm\bigl( \bmu, \bSigma \bigr),
		\end{equation*}
		wobei die Parameter $\bmu$ und $\bSigma$ gegeben sind durch
		\begin{align*}
			\bmu
				&\coloneqq \bmu_0^\top \bphi(\newinst)
		\\[2mm]
			\bSigma
				&\coloneqq \bphi(\newinst)^\top \bSigma_0 \bphi(\newinst) + \sigma^2.
		\end{align*}
	\end{mytheo}

	\begin{mytask}[task:reg-prior-pred-dist]
		Machen Sie sich diese Aussage anhand von Satz \ref{th:stat-bayes-norm} und mit Aufgabe \ref{task:reg-bayes-reg-joint-dist} klar.
	\end{mytask}
\end{frame}


% Interpretation
\begin{frame}{Interpretation}{}
	\begin{itemize}
		\item Der Erwartungswert $\bmu$ ergibt sich als das Skalarprodukt des Erwartungswerts des Priors mit dem Featurevektor.
		\item Der Term $\bphi(\newinst)^\top \bSigma_0 \bphi(\newinst)$ in der Formel für $\bSigma$ spiegelt die Unsicherheit der
			Parameter $\btheta$ unter dem Prior wider.
		\item $\sigma^2$ stellt Unsicherheit aufgrund von Datenrauschen oder nicht modellierten Effekten dar.
		\item Durch jedes $\btheta$, das wir vom Prior sampeln, erhält man eine Funktion $f(\newinst) \coloneqq \btheta^\top \bphi(\newinst)$.
	\end{itemize}

	\begin{boxGray}
		Der Prior $p(\btheta)$ über $\btheta$ induziert eine Verteilung $p\bigl( f(\bx) \bigr)$ über mögliche Funktionen.
	\end{boxGray}
\end{frame}


% Beispiel
\begin{frame}{Beispiel}{}
	\begin{itemize}
		\item Wir nehmen an, dass die Labels durch die folgende (in der Praxis natürlich unbekannte) Funktion generiert werden
		\begin{equation*}
			f(x) = \sin(x).
		\end{equation*}
		\item Zusätzlich addieren wir einen Fehler-Term $\varepsilon$, für den $\varepsilon \sim \norm(0, \sigma^2)$ gilt.
		\item Ein Trainingsbeispiel $(x_n, y_n)$ hat dann die Form
		\begin{equation*}
			\bigl( x_n, y_n \bigr) = \bigl( x_n, \sin(x_n) + \varepsilon_n \bigr), \qquad n = 1, 2, \ldots, N.
		\end{equation*}
		\item Als Prior über die Parameter $\btheta$ wählen wir eine zentrierte (d.\,h. mit Erwartungswert $\bmu_0 = \0$),
			isotropische Normalverteilung (hier $\bSigma_0 = \bI$), also
		\begin{equation*}
			\btheta \sim \norm(\0, \bI).
		\end{equation*}
	\end{itemize}
\end{frame}

% Berücksichtigung der Trainingsdaten
\begin{frame}{Berücksichtigung der Trainingsdaten}{}
	\begin{itemize}
		\item Unser bisheriges Modell nutzt noch nicht die Information aus den Trainingsdaten.
		\item Statt dem Prior $p(\btheta)$ müssen wir den Posterior $p(\btheta \mid \bX, \by)$ benutzen.
		\item Hierfür benutzen wir wieder den \keyword{Satz von Bayes}
		\begin{equation}
			p\bigl( \btheta \mid \bX, \by \bigr)
				= \frac{\overbracket{
					p\bigl( \by \mid \bX, \btheta \bigr)
				}^{\text{Likelihood}} \cdot \overbracket{p\bigl( \btheta \bigr)}^{\text{Prior}}}{p\bigl( \by \mid \bX \bigr)}
		\end{equation}
	\end{itemize}
\end{frame}


% Berücksichtigung der Trainingsdaten (Forts.)
\begin{frame}{Berücksichtigung der Trainingsdaten (Forts.)}{}
	Der Posterior kann wieder analytisch berechnet werden:
	\begin{mytheo}[th:reg-parameter-posterior]
		Für den Posterior gilt $\btheta \mid \bX, \by \sim \norm\bigl( \bmu_N, \bSigma_N \bigr)$ mit
		\begin{align*}
			\bSigma_N
				&\coloneqq \bigl( \bSigma_0^{-1} + \sigma^{-2} \bPhi^\top \bPhi \bigr)^{-1}
		\\[2mm]
			\bmu_N
				&\coloneqq \bSigma_N \bigl( \bSigma_0^{-1} \bmu_0 + \sigma^{-2} \bPhi^\top \by \bigr).
		\end{align*}
	\end{mytheo}
	
	\begin{mytask}[task:reg-posterior-pred-dist]
		Machen Sie sich diese Aussage anhand von Satz \ref{th:stat-bayes-norm} klar. Zusätzlich überlegen Sie sich bitte, wie
		Sie dieses Ergebnis mit der Verbunddichte aus Aufgabe \ref{task:reg-bayes-reg-joint-dist} erhalten.
	\end{mytask}
\end{frame}


% Posterior Predictive Distribution
\begin{frame}{Posterior Predictive Distribution}{}
	\begin{itemize}
		\item Die Parameter der \underline{Posterior} Predictive Distribution werden analog zu Satz \ref{th:reg-prior-predictive-dist} bestimmt
			(vergleiche dazu Aufgabe \ref{task:reg-prior-pred-dist}).
		\item Man ersetzt dort lediglich $\bmu_0$ durch $\bmu_N$ und $\bSigma_0$ durch $\bSigma_N$.
		\item Hier nutzen wir die \keyword{Konjugiertheit} des Priors aus:
			Da der Posterior dieselbe funktionale Form hat wie der Prior, können wir Letzteren in Satz \ref{th:reg-prior-predictive-dist}
			einfach durch den Posterior ersetzen.
	\end{itemize}
	
	\begin{mytheo}[th:reg-posterior-predictive-dist]
		Für die Posterior Predictive Distribution gilt $y \mid \newinst \sim \norm\bigl( \bmu, \bSigma \bigr)$, wobei die Parameter
		$\bmu$ und $\bSigma$ gegeben sind durch
		\begin{align*}
			\bmu
				&\coloneqq \bmu_N^\top \bphi(\newinst)
		\\[2mm]
			\bSigma
				&\coloneqq \bphi(\newinst)^\top \bSigma_N \bphi(\newinst) + \sigma^2.
		\end{align*}
	\end{mytheo}
\end{frame}
