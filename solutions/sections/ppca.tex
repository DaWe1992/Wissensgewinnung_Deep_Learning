\section{Probabilistische PCA (PPCA)}

% SOLUTION
\subsection*{Lösung zu Aufgabe \ref{task:ppca-joint-dist}}

Es gilt
\begin{align*}
	\bSigma_{\bx\bx}
		&= \E\bigl[ (\bx - \E[\bx]) (\bx - \E[\bx])^\top \bigr]
\\[2mm]
		&= \E\bigl[ (\bW \bz + \bmu + \beps - \bmu) (\bW \bz + \bmu + \beps - \bmu)^\top \bigr]
\\[2mm]
		&= \E\bigl[ (\bW \bz + \beps) (\bW \bz + \beps)^\top \bigr]
\\[2mm]
		&= \E\bigl[ \bW \bz \bz^\top \bW^\top + \bW \bz \beps^\top + \beps \bz^\top \bW^\top + \beps \beps^\top \bigr]
\\[2mm]
		&= \E\bigl[ \bW \bz \bz^\top \bW^\top \bigr] + \E\bigl[ \beps \beps^\top \bigr]
\\[2mm]
		&= \bW \bW^\top + \sigma^2 \bI.
\end{align*}


% SOLUTION
\subsection*{Lösung zu Aufgabe \ref{task:ppca-posterior-covariance}}

Für die Kovarianz des Posteriors errechnet man
\begin{align*}
	\bSigma_{\bz\,\mid\,\bx}
		&= \bSigma_{\bz\bz} - \bSigma_{\bz\bx} \bSigma_{\bx\bx}^{-1} \bSigma_{\bx\bz}
\\[2mm]
		&= \bI - \bW^\top \bC^{-1} \bW
\\[2mm]
		&= \bI - \bM^{-1} \bW^\top \bW
\\[2mm]
		&= \bI - \bM^{-1} \bigl( \bM - \sigma^2 \bI \bigr)
\\[2mm]
		&= \bI - \bM^{-1} \bM + \sigma^2 \bM^{-1}
\\[2mm]
		&= \sigma^2 \bM^{-1}.
\end{align*}


% SOLUTION
\subsection*{Lösung zu Aufgabe \ref{task:ppca-m-step-derivative-sigma-squared}}

Für die Ableitung nach $\sigma^2$ betrachten wir die Zielfunktion $f(\sigma^2)$, wobei wir das Ergebnis $\bmu = \bxmean$ wieder direkt einsetzen
\begin{equation*}
    	\begin{aligned}
        		f(\sigma^2) = -\sum_{n=1}^N \Biggl\{ 
            		&\frac{M}{2} \ln\bigl( 2 \pi \sigma^2 \bigr) + \frac{1}{2 \sigma^2} \Vert \bx_n - \bxmean \Vert^2
		\\
            		&- \frac{1}{\sigma^2} \E[\bz_n]^\top \bW^\top (\bx_n - \bxmean) + \frac{1}{2 \sigma^2} \trace\Bigl( \E\bigl[ \bz_n \bz_n^\top \bigr] 				\bW^\top \bW \Bigr) 
        		\Biggr\}
    \end{aligned}
\end{equation*}
Wir bilden zunächst die partiellen Ableitungen der einzelnen Terme innerhalb der äußersten Klammer nach der Variablen $\sigma^2$:
\begin{align*}
	\frac{\partial}{\partial \sigma^2} \left( \frac{M}{2} \ln(2 \pi \sigma^2) \right) 
        		&= \frac{M}{2 \sigma^2}
\\[2mm]
	\frac{\partial}{\partial \sigma^2} \left( \frac{1}{2 \sigma^2} \Vert \bx_n - \bxmean \Vert^2 \right) 
        		&= -\frac{1}{2 (\sigma^2)^2} \Vert \bx_n - \bxmean \Vert^2
\\[2mm]
	\frac{\partial}{\partial \sigma^2} \left( -\frac{1}{\sigma^2} \E[\bz_n]^\top \bW^\top (\bx_n - \bxmean) \right) 
        		&= \frac{1}{(\sigma^2)^2} \E[\bz_n]^\top \bW^\top (\bx_n - \bxmean)
\\[2mm]
	\frac{\partial}{\partial \sigma^2} \left( \frac{1}{2 \sigma^2} \trace\Bigl( \E\bigl[ \bz_n \bz_n^\top \bigr] \bW^\top \bW \Bigr) \right) 
		&= -\frac{1}{2 (\sigma^2)^2} \trace\Bigl( \E\bigl[ \bz_n \bz_n^\top \bigr] \bW^\top \bW \Bigr)
\end{align*}
Das Einsetzen der abgeleiteten Terme zurück in die Summe liefert
\begin{align*}
	\frac{\partial}{\partial \sigma^2} f\bigl( \sigma^2 \bigr)
		&= \begin{aligned}[t]
            		-\sum_{n=1}^N \Biggl\{ 
                			&\frac{M}{2 \sigma^2} - \frac{1}{2 (\sigma^2)^2} \Vert \bx_n - \bxmean \Vert^2
		\\
                			&+ \frac{1}{(\sigma^2)^2} \E[\bz_n]^\top \bW^\top (\bx_n - \bxmean)
					- \frac{1}{2 (\sigma^2)^2} \trace\bigl( \E\bigl[ \bz_n \bz_n^\top \bigr] \bW^\top \bW \bigr) 
            		\Biggr\} 
        		\end{aligned}
\\[3mm]
		&= \begin{aligned}[t]
            		-\frac{N M}{2 \sigma^2} - \sum_{n=1}^N \Biggl\{ 
                			&-\frac{1}{2 (\sigma^2)^2} \Vert \bx_n - \bxmean \Vert^2 + \frac{1}{(\sigma^2)^2} \E[\bz_n]^\top \bW^\top (\bx_n - \bxmean)
		\\
                			&- \frac{1}{2 (\sigma^2)^2} \trace\bigl( \E\bigl[ \bz_n \bz_n^\top \bigr] \bW^\top \bW \bigr) 
            		\Biggr\}.
        \end{aligned}
\end{align*}
Indem wir nun den gemeinsamen Faktor $\frac{1}{2(\sigma^2)^2}$ aus dem verbleibenden Summenausdruck ausklammern, drehen sich aufgrund des Minuszeichens vor der Summe alle Vorzeichen im Inneren um. Damit ergibt sich die gesuchte Identität:
\begin{equation*}
    	\begin{split}
        		\frac{\partial}{\partial \sigma^2} f\bigl( \sigma^2 \bigr)
			= -\frac{N M}{2 \sigma^2} + \frac{1}{2 \bigl( \sigma^2 \bigr)^2} \sum_{n=1}^N \biggl\{
                		&\Vert \bx_n - \bxmean \Vert^2 - 2 \E[\bz_n]^\top \bW^\top (\bx_n - \bxmean)
	\\
                		&+ \trace\bigl( \E\bigl[ \bz_n \bz_n^\top \bigr] \bW^\top \bW \bigr)
            	\biggr\}.
	\end{split}
\end{equation*}
